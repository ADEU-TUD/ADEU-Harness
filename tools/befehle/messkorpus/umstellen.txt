# Einstellungen umstellen (stell:*), mit erwartetem Wert
== prüfteil ==
mach mal dunkel => stell:theme ; wert=DARK
bitte auf hell stellen => stell:theme ; wert=LIGHT
switch to dark mode => stell:theme ; wert=DARK
design wie das system => stell:theme ; wert=SYSTEM
dunkelmodus aus => stell:theme ; wert=LIGHT
schrift größer => stell:text_size|ort:option.appearance.textsize ; wert=1.15
mach die schrift bitte sehr groß => stell:text_size ; wert=1.3
schriftgröße wieder normal => stell:text_size ; wert=1.0
stell die app auf englisch => stell:language ; wert=en
sprache deutsch => stell:language ; wert=de
lora ausblenden => stell:lora|ort:option.lora.mode ; wert=OFF
lora nur im chat zeigen => stell:lora ; wert=CHAT
lora wieder überall anzeigen => stell:lora ; wert=EVERYWHERE
pseudonymisierung einschalten => stell:pseudonymize ; wert=true
schalte das ersetzen von namen ab => stell:pseudonymize ; wert=false
websuche an => stell:web ; wert=true
keine websuche mehr => stell:web ; wert=false
dokumente erstellen ausschalten => stell:documents ; wert=false
das modell soll automatisch gewählt werden => stell:auto_model ; wert=true
automatische modellwahl aus => stell:auto_model ; wert=false
arbeitsschritte bitte immer einklappen => stell:collapse_steps ; wert=true
startanimation aus => stell:start_animation ; wert=false
lora soll ruhiger sein => stell:lora_calm ; wert=true
lora soll nur noch lokal arbeiten => stell:lora_place ; wert=LOCAL
stell lora auf nur server => stell:lora_place ; wert=SERVER
lora wieder gemischt => stell:lora_place ; wert=MIXED
turn off the start animation => stell:start_animation ; wert=false
